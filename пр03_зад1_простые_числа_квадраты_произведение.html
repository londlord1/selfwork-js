// Исходные данные const numbers = [12, 7, 23, 45, 18, 31, 6, 2, 19, 8]; /** * Функция проверки: является ли число простым * Простое число — это натуральное число больше 1, которое имеет только два делителя: 1 и само себя. */ function isPrime(num) { if (num <= 1) return false; // 1 и меньше — не простые if (num === 2) return true; // 2 — простое if (num % 2 === 0) return false; // Четные числа, кроме 2, не простые // Проверяем делители от 3 до квадратного корня из числа (только нечетные) for (let i = 3; i <= Math.sqrt(num); i += 2) { if (num % i === 0) return false; } return true; } /** * 1. getPrimes() — возвращает только простые числа из массива */ function getPrimes(arr) { return arr.filter(isPrime); } /** * 2. getSquares() — возвращает квадраты всех чисел массива */ function getSquares(arr) { return arr.map(num => num * num); } /** * 3. getProduct() — возвращает произведение всех чисел через reduce */ function getProduct(arr) { // Важно: начальное значение должно быть 1, иначе умножение сломается return arr.reduce((accumulator, currentValue) => accumulator * currentValue, 1); } // --- Тестирование --- console.log("Исходный массив:", numbers); console.log("Простые числа:", getPrimes(numbers)); // Ожидаемый результат: [7, 23, 31, 2, 19] console.log("Квадраты чисел:", getSquares(numbers)); // Ожидаемый результат: [144, 49, 529, 2025, 324, 961, 36, 4, 361, 64] console.log("Произведение всех чисел:", getProduct(numbers)); // Ожидаемый результат: очень большое число (12 * 7 * 23 * ... * 8)

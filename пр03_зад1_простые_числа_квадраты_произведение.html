ПР03 • Зад.1

Простые числа, квадраты, произведение

Функции getPrimes(), getSquares() и getProduct() для массива чисел. getProduct() реализован через reduce с начальным значением 1; isPrime() корректно обрабатывает 0, 1, отрицательные и дробные числа.

Исходный массив

Числа через запятую или пробел. Можно проверить крайние случаи: 1, 0, −5, 2.

function isPrime(num) {
  if (!Number.isInteger(num) || num < 2) return false;
  if (num === 2) return true;
  if (num % 2 === 0) return false;
  for (let i = 3; i * i <= num; i += 2) {
    if (num % i === 0) return false;
  }
  return true;
}

function getPrimes(arr) {
  return arr.filter(isPrime);
}

function getSquares(arr) {
  return arr.map(num => num * num);
}

function getProduct(arr) {
  return arr.reduce((acc, current) => acc * current, 1);
}

function parseArray(text) {
  return text.split(/[,\s;]+/).filter(Boolean).map(Number).filter(n => Number.isFinite(n));
}

function fmt(list) {
  return '[' + list.join(', ') + ']';
}

function runAll() {
  const numbers = parseArray($('arr').value);
  if (numbers.length === 0) {
    $('result').innerHTML = '';
    out('<span class="err">Введите хотя бы одно число</span>');
    return;
  }

  const primes = getPrimes(numbers);
  const squares = getSquares(numbers);
  const product = getProduct(numbers);
  const invalid = $('arr').value.split(/[,\s;]+/).filter(Boolean).length - numbers.length;

  $('result').innerHTML = `
    <div class="item"><span>Исходный массив</span><b>${fmt(numbers)}</b></div>
    <div class="item"><span>getPrimes()</span><b>${fmt(primes)}</b></div>
    <div class="item"><span>getSquares()</span><b>${fmt(squares)}</b></div>
    <div class="item"><span>getProduct()</span><b>${product.toLocaleString('ru-RU')}</b></div>
    <div class="item"><span>Проверка isPrime()</span><b>${numbers.map(n => n + ' → ' + isPrime(n)).join(', ')}</b></div>`;

  out(`Простых чисел: <span class="hl">${primes.length}</span> из ${numbers.length}
Произведение через reduce с начальным значением 1: <span class="ok">${product.toLocaleString('ru-RU')}</span>
${invalid ? '<span class="err">Пропущено нечисловых значений: ' + invalid + '</span>' : ''}`);
}

function resetArr() {
  $('arr').value = '12, 7, 23, 45, 18, 31, 6, 2, 19, 8';
  runAll();
}

runAll();
